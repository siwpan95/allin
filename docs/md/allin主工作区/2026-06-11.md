```

├── main.py              # 入口 (v2.0 重写, 完整交互+音效)
├── constants.py          # 全局常量 (保留root级)
├── 启动游戏.bat

│   ├── card_engine.py    # 扑克牌组/牌型判定

├── systems/              # 游戏系统
│   ├── chess_pieces.py   # 30棋子+6羁绊
│   ├── tournament.py     # 瑞士轮+商店
│   └── events.py         # 事件/装备/诅咒
├── ui/                   # 界面
│   ├── components.py     # Button/Card/PieceCard/Bullet/LifeBar
│   └── renderer.py       # 旧渲染器(保留)
├── config/               # 配置
│   ├── *.xlsx            # 4个Excel源文件

│   └── config_loader.py  # Excel↔JSON双向转换
└── audio/                # 音频
    └── sound_manager.py  # pygame.mixer音效管理
```
